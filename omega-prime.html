/- ============================================================ -/
/- ## SECTION 2: OPTICAL PROPAGATION AND REFLECTION            -/
/- ============================================================ -/

/-- The speed of light c in vacuum (m/s, normalized to 1 in geometric units). -/
noncomputable def c : ℝ := 1

/-- Photon propagation: photon p travels from source to destination. -/
def photonReaches (p : Photon) (dest : EuclideanSpace ℝ (Fin 3)) : Prop :=
  ∃ t > p.timestamp, p.origin + (t - p.timestamp) • p.direction = dest

/-- First-pass absorption: photon hits retina and is absorbed. -/
def firstPassAbsorb (sys : TapetalSystem) (p : Photon) : Prop :=
  photonReaches p sys.W.retina ∧ p.wavelength ∈ sys.T.sensitivityRange

/-- Tapetal reflection: photon reflects back along incident vector. -/
def tapetalReflect (sys : TapetalSystem) (p : Photon) : Photon :=
  { p with
    direction := -p.direction,  -- reverse direction
    origin := sys.W.retina,
    timestamp := p.timestamp + dist p.origin sys.W.retina / c
  }

/-- Second-pass absorption: reflected photon hits retina again. -/
def secondPassAbsorb (sys : TapetalSystem) (p : Photon) : Prop :=
  let p' := tapetalReflect sys p
  photonReaches p' sys.W.retina

  if p.wavelength ∈ sys.T.sensitivityRange then
    let p1 := if firstPassAbsorb sys p then 1 else 0
    let p2 := if secondPassAbsorb sys p then sys.T.reflectivity else 0
    min 1 (p1 + p2)  -- probability capped at 1
  else 0

    (h_range : p.wavelength ∈ sys.T.sensitivityRange)

    absorptionProb sys p ≥ 2 * (1 - sys.T.reflectivity) * absorptionProb sys p := by
  simp [absorptionProb, h_range]
